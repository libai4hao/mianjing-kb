## 5. 统计 SQL（可直接跑，只读）

```sql
-- 0) 一律只读打开
-- sqlite3.connect("file:<path>?mode=ro", uri=True)

-- 1) 台账分布（kb.sqlite）
SELECT status, COUNT(*) FROM source_documents GROUP BY 1 ORDER BY 2 DESC;   -- processed 1392 / superseded 103 / rejected 20

-- 2) 题/簇（kb.sqlite）
SELECT COUNT(*) FROM questions;                        -- 20,984
SELECT COUNT(DISTINCT q_norm_hash) FROM questions;     -- 11,994
SELECT COUNT(DISTINCT doc_hash) FROM questions;         -- 1,212
SELECT COUNT(*) FROM canonical_questions;               -- 11,520
SELECT COUNT(*) FROM canonical_questions WHERE witness>=2; -- 3,684
SELECT COUNT(*) FROM questions WHERE COALESCE(a_text,'')='';        -- 5,699
SELECT COUNT(*) FROM questions WHERE LENGTH(COALESCE(a_text,''))<20; -- 6,557
SELECT COUNT(*) FROM (SELECT doc_hash,q_norm_hash,COUNT(*) c
  FROM questions GROUP BY 1,2 HAVING c>1);               -- 852

-- 2b) 见证口径（Q-2）：distinct collection，空 collection 回退 doc_hash
SELECT COUNT(DISTINCT COALESCE(NULLIF(TRIM(sd.collection), ''), 'doc:' || q.doc_hash))
  FROM questions q LEFT JOIN source_documents sd ON sd.doc_hash = q.doc_hash;  -- 152
SELECT COUNT(DISTINCT sd.collection) FROM source_documents
  WHERE collection IS NOT NULL AND collection <> '';                            -- 153
SELECT COUNT(*) FROM source_documents WHERE collection IS NULL OR collection=''; -- 63

-- 3) 产物台账（015）
SELECT text_source, COUNT(*), SUM(questions_n) FROM document_products GROUP BY 1; -- md 1296/19625, orphan 74/335, source 96/1024
SELECT COUNT(*) FROM document_products WHERE questions_n=0;                       -- 254

-- 4) 索引一致性（G4 口径；当前检索库仍是旧库，不可直接对新题集比较）
SELECT (SELECT COUNT(*) FROM questions) AS q,
       (SELECT COUNT(*) FROM vec_questions) AS vec,
       (SELECT COUNT(*) FROM questions_fts) AS fts;
```

注意：G4 的 SQL **不能**证明"向量与题目同源"（`docs/27` P1-2），需要内容抽样才有意义。
